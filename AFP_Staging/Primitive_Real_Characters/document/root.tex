\title{Primitive Real Dirichlet Characters and Fundamental Discriminants}
\author{Lawrence C. Paulson}
\maketitle
\begin{abstract}
We classify primitive real Dirichlet characters by their prime-power components and construct the associated fundamental discriminant using quadratic reciprocity. The resulting character formula is stated independently of applications to Green's proof of the nonexistence of Siegel zeros.
\end{abstract}
